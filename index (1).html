-- Script para Free Fire Max e Free Fire normal function headshot()     local x, y = getScreenSize()     local headshotX = x / 2     local headshotY = y / 2 - 50     touchDown(headshotX, headshotY)     sleep(10)     touchUp(headshotX, headshotY) end function melhorarMira()     local x, y = getScreenSize()     local miraX = x / 2     local miraY = y / 2     touchDown(miraX, miraY)     sleep(10)     touchUp(miraX, miraY) end function jogoAberto()     local packageName = "com.dts.freefireth"     return isAppRunning(packageName) end function main()     while true do         if jogoAberto() then             headshot()             melhorarMira()             optimizeGame()             checkAntiban()             checkAntiblacklist()         end         sleep(100)     end end function optimizeGame()     local cpuUsage = getCpuUsage()     if cpuUsage > 50 then         setCpuUsage(30)     end     local ramUsage = getRamUsage()     if ramUsage > 500 then         setRamUsage(300)     end end function checkAntiban()     local banStatus = getBanStatus()     if banStatus == "banido" then         alert("Você está banido!")     end end function checkAntiblacklist()     local blacklisted = isBlacklisted()     if blacklisted then         alert("Você está na blacklist!")     end end main()
